<script>

let db=JSON.parse(localStorage.getItem("clientes")||"[]");

function save(){
localStorage.setItem("clientes",JSON.stringify(db));
mostrarLista();
}

function login(){
let u=user.value;
let p=pass.value;
let c=db.find(x=>x.user==u && x.pass==p);
if(!c){alert("Datos incorrectos");return;}
clientePanel.style.display="block";
clientePanel.innerHTML=`
<div class="item">
Servicio: ${c.servicio}<br>
Tipo: ${c.tipo}<br>
Vence en: ${c.vence} días<br>
<a href="https://wa.me/56999225842">Soporte WhatsApp</a>
</div>`;
}

function adminLogin(){
let clave=prompt("Clave admin");
if(clave==="edicson123"){
admin.style.display="block";
mostrarLista();
}
}

function crear(){
db.push({
user:nUser.value,
pass:nPass.value,
servicio:servicio.value,
tipo:tipo.value,
vence:vence.value
});
save();
}

function mostrarLista(){
lista.innerHTML="";
db.forEach((c,i)=>{
lista.innerHTML+=`
<div class="item">
${c.user} — ${c.servicio}
<button onclick="delCli(${i})">Eliminar</button>
</div>`;
});
}

function delCli(i){
db.splice(i,1);
save();
}

</script>
